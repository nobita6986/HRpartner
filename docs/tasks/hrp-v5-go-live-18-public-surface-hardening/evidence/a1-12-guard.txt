119	export async function evaluateRateLimits(input: EnforceRateLimitsInput): Promise<RateLimitOutcome> {
120	  const env = input.env ?? process.env;
121	  const requestId = input.requestId ?? null;
122	
123	  let runtime: ReturnType<typeof getRateLimitRuntime>;
124	  try {
125	    runtime = getRateLimitRuntime(env);
126	  } catch (err) {
127	    logUnavailable(input.routeClass, requestId, err);
128	    return { kind: 'unavailable' };
129	  }
130	
131	  for (const bucket of input.buckets) {
132	    const canonical = canonicalValueFor(bucket.rule, bucket.value);
133	    const rule = effectiveRule(bucket.rule, canonical);
134	    const identifier = hashRateLimitIdentifier(rule, canonical, runtime.hashSecret);
135	
136	    let decision: RateLimitDecision;
137	    try {
138	      decision = await runtime.provider.limit(rule, identifier);
139	    } catch (err) {
140	      logUnavailable(input.routeClass, requestId, err, rule.surface);
141	      return { kind: 'unavailable' };
142	    }
143	
144	    if (!decision.allowed) {
145	      warn('rate_limit.denied', requestId, {
146	        route: input.routeClass,
147	        status: 429,
148	        outcome: 'rate_limited',

150	      });
151	      return { kind: 'rate-limited', decision };
152	    }

155	  return { kind: 'allowed' };

165	  const outcome = await evaluateRateLimits(input);
